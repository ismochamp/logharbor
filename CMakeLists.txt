cmake_minimum_required(VERSION 3.16)
project(LogHarbor LANGUAGES CXX)
add_executable(logharbor src/logharbor.cpp)
target_compile_features(logharbor PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(logharbor PRIVATE /W4 /utf-8)
else()
  target_compile_options(logharbor PRIVATE -Wall -Wextra -Wpedantic)
endif()
